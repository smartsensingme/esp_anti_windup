cmake_minimum_required(VERSION 3.16)
if(ESP_PLATFORM)
    idf_component_register(SRCS "anti_windup.c" INCLUDE_DIRS "include")
    return()
endif()
project(anti_windup VERSION 0.1.0 LANGUAGES C)
add_library(anti_windup anti_windup.c)
target_include_directories(anti_windup PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(anti_windup PUBLIC c_std_99)
if(NOT MSVC)
    target_link_libraries(anti_windup PUBLIC m)
endif()
option(ANTI_WINDUP_BUILD_TESTS "Build host tests" OFF)
option(ANTI_WINDUP_BUILD_EXAMPLES "Build portable example" OFF)
if(ANTI_WINDUP_BUILD_TESTS)
    enable_testing()
    add_executable(test_anti_windup tests/test_anti_windup.c)
    target_link_libraries(test_anti_windup PRIVATE anti_windup)
    if(NOT MSVC)
        target_compile_options(anti_windup PRIVATE -Wall -Wextra -Werror -pedantic)
        target_compile_options(test_anti_windup PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
    endif()
    add_test(NAME anti_windup COMMAND test_anti_windup)
endif()
if(ANTI_WINDUP_BUILD_EXAMPLES)
    add_executable(parallel_integrators examples/parallel_integrators.c)
    target_link_libraries(parallel_integrators PRIVATE anti_windup)
endif()
